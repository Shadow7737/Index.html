// --- Web Audio Synthesizer (No External Audio Assets Needed) ---
class SoundFX {
constructor() {
this.ctx = null;
}
init() {
if (!this.ctx) {
this.ctx = new (window.AudioContext || window.webkitAudioContext)();
}
}
playCoin() {
if (!this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sine';
osc.frequency.setValueAtTime(987.77, this.ctx.currentTime); // B5
osc.frequency.exponentialRampToValueAtTime(1318.51, this.ctx.currentTime + 0.1); // E6
gain.gain.setValueAtTime(0.15, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.12);
osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.12);
}
playJump() {
if (!this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'triangle';
osc.frequency.setValueAtTime(220, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(600, this.ctx.currentTime + 0.2);
gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.2);
osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.2);
}
playSlide() {
if (!this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(300, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(100, this.ctx.currentTime + 0.18);
gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.18);
osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.18);
}
playCrash() {
if (!this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(150, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(40, this.ctx.currentTime + 0.4);
gain.gain.setValueAtTime(0.4, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.4);
osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.4);
}
playPowerup() {
if (!this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sine';
osc.frequency.setValueAtTime(440, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(880, this.ctx.currentTime + 0.25);
gain.gain.setValueAtTime(0.25, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.25);
osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.25);
}

// --- Spawning Obstacles & Collectibles ---
function spawnPattern() {
const zSpawn = -55;
const lanePool = [0, 1, 2];

// Choose 1 or 2 lanes for obstacles
const numObstacles = Math.random() < 0.4 ? 2 : 1;
const selectedLanes = [];

for (let i = 0; i < numObstacles; i++) {
const index = Math.floor(Math.random() * lanePool.length);
selectedLanes.push(lanePool.splice(index, 1)[0]);
}

selectedLanes.forEach(laneIdx => {
const randType = Math.random();
if (randType < 0.45) {
spawnTrain(LANES[laneIdx], zSpawn);
} else if (randType < 0.75) {
spawnLowHurdle(LANES[laneIdx], zSpawn);
} else {
spawnHighHurdle(LANES[laneIdx], zSpawn);
}
});

// Spawn Coins in open lane(s)
if (lanePool.length > 0 && Math.random() < 0.8) {
const coinLane = lanePool[Math.floor(Math.random() * lanePool.length)];
for (let c = 0; c < 4; c++) {
spawnCoin(LANES[coinLane], zSpawn - (c * 2.2));
}
}

// Occasional Powerup Spawn
if (Math.random() < 0.15) {
const powerLane = Math.floor(Math.random() * 3);
spawnPowerup(LANES[powerLane], zSpawn - 10);
}
}

// --- Controls Logic (Swipe + Buttons + Keyboard) ---
function moveLeft() {
if (currentLane > 0) {
currentLane--;
targetX = LANES[currentLane];
}
}

function moveRight() {
if (currentLane < 2) {
currentLane++;
targetX = LANES[currentLane];
}
}

function jump() {
if (!isJumping) {
isJumping = true;
playerVelY = JUMP_FORCE;
sfx.playJump();
}
}

function slide() {
if (!isSliding) {
isSliding = true;
slideTimer = 35; // Frames duration
sfx.playSlide();

// Fast drop if jumping
if (isJumping) {
playerVelY = -0.3;
}
}
}

// Touch Swipe Gesture Detection
let touchStartX = 0;
let touchStartY = 0;

function setupInputListeners() {
// On-screen Touch Buttons
document.getElementById('btn-left').addEventListener('touchstart', (e) => { e.preventDefault(); moveLeft(); });
document.getElementById('btn-right').addEventListener('touchstart', (e) => { e.preventDefault(); moveRight(); });
document.getElementById('btn-jump').addEventListener('touchstart', (e) => { e.preventDefault(); jump(); });
document.getElementById('btn-slide').addEventListener('touchstart', (e) => { e.preventDefault(); slide(); });

// Click support for touch buttons (fallback)
document.getElementById('btn-left').addEventListener('click', moveLeft);
document.getElementById('btn-right').addEventListener('click', moveRight);
document.getElementById('btn-jump').addEventListener('click', jump);
document.getElementById('btn-slide').addEventListener('click', slide);

// Screen Swipe Handlers
window.addEventListener('touchstart', (e) => {
sfx.init(); // Initialize Web Audio context on first touch
if (e.touches.length > 0) {
touchStartX = e.touches[0].clientX;
touchStartY = e.touches[0].clientY;
}
}, { passive: true });

window.addEventListener('touchend', (e) => {
if (!e.changedTouches || e.changedTouches.length === 0) return;
const dx = e.changedTouches[0].clientX - touchStartX;
const dy = e.changedTouches[0].clientY - touchStartY;

const absX = Math.abs(dx);
const absY = Math.abs(dy);

if (Math.max(absX, absY) > 30) { // Threshold
if (absX > absY) {
// Horizontal Swipe
if (dx > 0) moveRight();
else moveLeft();
} else {
// Vertical Swipe
if (dy < 0) jump();
else slide();
}
}
}, { passive: true });

// Keyboard Fallback
window.addEventListener('keydown', (e) => {
sfx.init();
if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') moveLeft();
if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') moveRight();
if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W' || e.key === ' ') jump();
if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') slide();
});

// Start / Restart Button
btnStart.addEventListener('click', () => {
sfx.init();
startRun();
});
}

// --- Core Game Loop & Updates ---
let patternTimer = 0;

function updateGame() {
if (!isPlaying) return;

// Increase difficulty over time
gameSpeed += 0.0001;

// 1. Player X-axis lerp to lane
playerMesh.position.x += (targetX - playerMesh.position.x) * 0.22;

// 2. Player Jump / Gravity Y-axis
if (isJumping) {
playerY += playerVelY;
playerVelY += GRAVITY;
if (playerY <= GROUND_Y) {
playerY = GROUND_Y;
playerVelY = 0;
isJumping = false;
}
}

// 3. Player Slide Logic
if (isSliding) {
slideTimer--;
playerMesh.scale.set(1, 0.5, 1); // Compress character
if (slideTimer <= 0) {
isSliding = false;
playerMesh.scale.set(1, 1, 1);
}
} else {
playerMesh.scale.set(1, 1, 1);
}

playerMesh.position.y = playerY;

// Player running animation tilt
playerMesh.rotation.z = (playerMesh.position.x - targetX) * 0.12;

// 4. Track Decorations recycling
trackDecorations.forEach(arch => {
arch.position.z += gameSpeed;
if (arch.position.z > 8) {
arch.position.z -= 90;
}
});

// 5. Spawn Logic
patternTimer += gameSpeed;
if (patternTimer > 16) {
patternTimer = 0;
spawnPattern();
}

// 6. Update & Check Obstacles
for (let i = obstacles.length - 1; i >= 0; i--) {
const obs = obstacles[i];
obs.position.z += gameSpeed;

// Collision Check with Player Box
const pBox = {
minX: playerMesh.position.x - 0.35,
maxX: playerMesh.position.x + 0.35,
minY: playerY,
maxY: playerY + (isSliding ? 0.7 : 1.6),
minZ: -0.4,
maxZ: 0.4
};

const data = obs.userData;
const oBox = {
minX: obs.position.x - (data.width / 2),
maxX: obs.position.x + (data.width / 2),
minY: obs.userData.minCapY || 0,
maxY: obs.userData.minCapY ? 3.0 : data.height,
minZ: obs.position.z - (data.length / 2),
maxZ: obs.position.z + (data.length / 2)
};

// Collision Detection Test
if (
pBox.maxX > oBox.minX && pBox.minX < oBox.maxX &&
pBox.maxY > oBox.minY && pBox.minY < oBox.maxY &&
pBox.maxZ > oBox.minZ && pBox.minZ < oBox.maxZ
) {
gameOver();
return;
}

// Cleanup passed obstacles
if (obs.position.z > 10) {
scene.remove(obs);
obstacles.splice(i, 1);
}
}

// 7. Update Coins
for (let i = coins.length - 1; i >= 0; i--) {
const coin = coins[i];
coin.position.z += gameSpeed;
coin.rotation.z += 0.05;

// Magnet Powerup pull effect
if (activePowerup === 'magnet' && coin.position.z > -25) {
coin.position.x += (playerMesh.position.x - coin.position.x) * 0.25;
coin.position.y += (playerY + 0.8 - coin.position.y) * 0.25;
}

// Collect Check
const distSq = Math.pow(playerMesh.position.x - coin.position.x, 2) +
Math.pow((playerY + 0.8) - coin.position.y, 2) +
Math.pow(0 - coin.position.z, 2);

if (distSq < 1.2) {
sfx.playCoin();
const value = activePowerup === 'multiplier' ? 2 : 1;
coinsCollected += value;
coinsDisplay.innerText = coinsCollected;

scene.remove(coin);
coins.splice(i, 1);
continue;
}

if (coin.position.z > 10) {
scene.remove(coin);
coins.splice(i, 1);
}
}

// 8. Update Powerups
for (let i = powerups.length - 1; i >= 0; i--) {
const p = powerups[i];
p.position.z += gameSpeed;
p.rotation.y += 0.05;

const distSq = Math.pow(playerMesh.position.x - p.position.x, 2) +
Math.pow(0 - p.position.z, 2);

if (distSq < 1.4) {
sfx.playPowerup();
activatePowerup(p.userData.type);
scene.remove(p);
powerups.splice(i, 1);
continue;
}

if (p.position.z > 10) {
scene.remove(p);
powerups.splice(i, 1);
}
}

// 9. Score Calculation
distanceRun += gameSpeed * 0.5;
score = Math.floor(distanceRun);
scoreDisplay.innerText = score;

// Powerup Timer Update
if (activePowerup) {
powerupTimeLeft -= 0.016;
const pct = Math.max(0, (powerupTimeLeft / 7) * 100);
document.getElementById('powerup-progress').style.width = pct + '%';

if (powerupTimeLeft <= 0) {
activePowerup = null;
document.getElementById('powerup-bar-container').classList.add('hidden');
document.getElementById('multiplier-badge').classList.add('hidden');
}
}
}

function activatePowerup(type) {
activePowerup = type;
powerupTimeLeft = 7.0; // Seconds

const barContainer = document.getElementById('powerup-bar-container');
const title = document.getElementById('powerup-title');
const badge = document.getElementById('multiplier-badge');

barContainer.classList.remove('hidden');

if (type === 'magnet') {
title.innerText = "COIN MAGNET ACTIVE";
title.className = "text-xs font-bold uppercase tracking-wider text-pink-400";
badge.classList.add('hidden');
} else if (type === 'multiplier') {
title.innerText = "2X COINS ACTIVE";
title.className = "text-xs font-bold uppercase tracking-wider text-emerald-400";
badge.classList.remove('hidden');
}
}

function startRun() {
// Reset State
obstacles.forEach(o => scene.remove(o));
coins.forEach(c => scene.remove(c));
powerups.forEach(p => scene.remove(p));
obstacles = [];
coins = [];
powerups = [];

currentLane = 1;
targetX = LANES[1];
playerY = GROUND_Y;
playerVelY = 0;
isJumping = false;
isSliding = false;

gameSpeed = 0.4;
distanceRun = 0;
score = 0;
coinsCollected = 0;
activePowerup = null;

scoreDisplay.innerText = "0";
coinsDisplay.innerText = "0";
document.getElementById('powerup-bar-container').classList.add('hidden');
document.getElementById('multiplier-badge').classList.add('hidden');

modalScreen.classList.add('hidden');
isPlaying = true;
}

function gameOver() {
isPlaying = false;
sfx.playCrash();

modalTitle.innerText = "GAME OVER";
modalSubtitle.innerText = "You crashed into an obstacle!";
finalScore.innerText = score;
finalCoins.innerText = coinsCollected;

statsContainer.classList.remove('hidden');
btnStart.innerText = "PLAY AGAIN";
modalScreen.classList.remove('hidden');
}

renderer.setSize(window.innerWidth, window.innerHeight);
}

requestAnimationFrame(animate);
updateGame();

}

};
